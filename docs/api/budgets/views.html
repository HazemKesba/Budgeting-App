budgets.views

Views for the budgets application.

Contains class-based views handling CRUD operations for user budgets. All views require authentication and delegate business logic to BudgetService.

  7from django.views import View
  8from django.shortcuts import render, redirect, get_object_or_404

 18@method_decorator(login_required, name='dispatch')
 19class BudgetListView(View):
 20    """Display a list of budgets for the authenticated user."""
 21    def get(self, request: HttpRequest) -> HttpResponse:
 22        """Handle GET request to retrieve and display user budgets.
 23
 24        Args:
 25            request: The current HTTP request.
 26
 27        Returns:
 28            Rendered budget list page.
 29        """
 30        budgets = BudgetService.get_user_budgets(request.user)
 31        return render(request, 'budgets/budget_list.html', {'budgets': budgets})

 34@method_decorator(login_required, name='dispatch')
 35class BudgetCreateView(View):
 36    """Handle creation of new budgets via form submission."""
 37    def get(self, request: HttpRequest) -> HttpResponse:
 38        """Render the budget creation form.
 39
 40        Args:
 41            request: The current HTTP request.
 42
 43        Returns:
 44            Rendered form page with 'Create' action label.
 45        """
 46        form = BudgetForm(request.user)
 47        return render(request, 'budgets/budget_form.html', {'form': form, 'action': 'Create'})
 48
 49    def post(self, request: HttpRequest) -> HttpResponse:
 50        """Validate and save a new budget.
 51
 52        Args:
 53            request: The current HTTP request containing form data.
 54
 55        Returns:
 56            Redirect to budget list on success, or re-rendered form with validation errors.
 57        """
 58        form = BudgetForm(request.user, request.POST)
 59        if form.is_valid():
 60            BudgetService.create_budget(request.user, form)
 61            messages.success(request, 'Budget created!')
 62            return redirect('budgets:budget_list')
 63        return render(request, 'budgets/budget_form.html', {'form': form, 'action': 'Create'})
